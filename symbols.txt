# Makaimura (MAME set 'makaimurg', Japan revision G) - the 6809 program's symbols
#
# Addresses and names only: no game data lives in this file.
# Format:  ADDRESS  KIND  NAME  [key=value ...]   [# comment]   (tools/symbols.py)
#   ADDRESS  hex 0000-FFFF; in the banked window 4000-5FFF, "B:XXXX" with its bank 0-4
#   KIND     func, code, data, var (RAM), table (count=N: 16-bit code addresses; bank=N: their targets' bank)
# The analysis (tools/disassemble.py) starts at the vectors (RESET 6000, IRQ 65C4); these lines name what it
# finds and add what it cannot work out alone.

# ======== the vectors ========
6000  func  reset            # ORCC #$50, sprite RAM filled with F8, 100 frames waited (SYNC), DP = 0, the layers
                             # cleared, bank 3, then JMP $553A with the return address in Y (607D)
65C4  func  irq              # the VBLANK: the sprite DMA (STB $3C00) first, the frame counter (<$20), the
                             # frame's jobs, then the game state's routine in bank 3; the bank put back from <$D9

# ======== the interrupt's dispatch ========
65BC  table  irq_states      count=4 bank=3   # by the game state <$02 (JSR [B,X] at 65F3, bank 3 selected)
65FB  func  irq_round_robin                   # one of four jobs a frame: <$21 & 3, its bank from 660D
6611  table  irq_round_robin_jobs  count=4    # (JMP [B,X] at 660B; banks 3, 4, 4, 4 from the table at 660D)
660D  data  irq_round_robin_banks             # 4 bytes: the bank of each job

# ======== the boot's continuation ========
607D  code  reset_after_ram_test   # where bank 3's RAM and ROM test (553A) returns: PSHS Y there, RTS
6134  code  main_loop_start        # the main loop's setup: the frame counters, the DSWs, sound off, then IRQs on
617F  code  main_loop              # the task queue at <$12: each word's high bit clear is a task (its number in
                                   # A: JSR through the table at 6188, in bank 3); 6160 clears it
6188  data  main_tasks             # (the words read by LDX A,X at 6174)

# ======== the objects ========
7E6D  func  objects_frame          # bank 4, then the objects' routines (8F0E, 9ACE, 9F0C)
9C97  table object_states_9AF4  count=4    # JSR [A,U] at 9AF4: U = 9C97 (gref --registers-at 9AF4), A 0-6; the words before it (9C8B-9C96) are pointers to data
4:4712 table object_handlers   count=19   # JSR [B,U] at 9F2D: by the object's type ($5,X); entry 15 is 0000 (unused)
4:470E table object_phase      count=2    # JSR [B,U] at 9F37: by the frame's parity (<$21 & 1)
